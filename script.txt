function save(){localStorage.setItem(KEY,JSON.stringify(items))}
function statusLabel(s){return {available:'Disponível',returning:'Em processo de devolução',returned:'Devolvido',discarded:'Descartado/encaminhado'}[s]||s}
function renderStats(){document.getElementById('total').textContent=items.length;document.getElementById('availableCount').textContent=items.filter(x=>x.status==='available').length;document.getElementById('returningCount').textContent=items.filter(x=>x.status==='returning').length;document.getElementById('returnedCount').textContent=items.filter(x=>x.status==='returned').length}
function renderMap(){const el=document.getElementById('armarios');el.innerHTML='';for(let c=1;c<=5;c++){let box=document.createElement('div');box.className='arm';box.innerHTML='<h3>Armário '+c+'</h3>';for(let s=7;s>=1;s--){let n=items.filter(x=>+x.cabinet===c&&+x.shelf===s).length;let sh=document.createElement('div');sh.className='shelf';sh.innerHTML='<span>Prat. function renderList(){let q=search.value.trim().toLowerCase();let arr=items.filter(x=>(!filter||(x.cabinet===filter.cabinet&&x.shelf===filter.shelf))&&(!q||[x.id,x.name,x.category,x.brand,x.color,x.description,`armário ${x.cabinet}`,`prateleira ${x.shelf}`,statusLabel(x.status)].join(' ').toLowerCase().includes(q)));const list=document.getElementById('itemList');list.innerHTML='';if(!arr.length){list.innerHTML='<div class="empty">Nenhum objeto encontrado.</div>';return}arr.sort((a,b)=>b.id.localeCompare(a.id));arr.forEach(x=>{let d=document.createElement('div');d.className='item';d.innerHTML=`<div class="thumb">${x.photo?`<img src="${x.photo}">`:'▧'}</div><div style="min-width:0;flex:1"><h3>${esc(x.name)} <span style="font-size:10px;color:#8a929b">${x.id}</span></h3><div class="meta">${esc(x.category)}${x.brand?' · '+esc(x.brand):''}${x.color?' · '+esc(x.color):''}<br>📍 Armário ${x.cabinet} → Prateleira ${x.shelf}<br>Encontrado em ${formatDate(x.date)}</div><span class="badge ${x.status}">${statusLabel(x.status)}</span><div style="display:flex;gap:6px;flex-wrap:wrap;margin-top:8px"><button onclick="event.stopPropagation();quickStatus('${x.id}','available')">Disponível</button><button onclick="event.stopPropagation();quickStatus('${x.id}','returning')">Em devolução</button><button onclick="event.stopPropagation();quickStatus('${x.id}','returned')">Devolvido</button><button onclick="event.stopPropagation();editItem('${x.id}')" class="primary">Editar</button></div></div>`;d.onclick=()=>showDetails(x);list.appendChild(d)})}
function showDetails(x){alert(`${x.id}\n\n${x.name}\nCategoria: ${x.category}\nMarca: ${x.brand||'—'}\nCor: ${x.color||'—'}\nLocal: Armário ${x.cabinet} → Prateleira ${x.shelf}\nData: ${formatDate(x.date)}\nStatus: ${statusLabel(x.status)}\n\nDescrição: ${x.description||'—'}\nObservações: ${x.notes||'—'}`)}
function openModal(id=null){editingId=id;const form=document.getElementById('form');form.reset();document.getElementById('preview').style.display='none';document.getElementById('modal').classList.add('show');document.getElementById('modalTitle').textContent=id?'Editar objeto':'Cadastrar objeto';document.getElementById('saveBtn').textContent=id?'Guardar alterações':'Salvar objeto';document.getElementById('deleteBtn').style.display=id?'inline-block':'none';if(id){const x=items.find(i=>i.id===id);if(!x)return;for(const k of ['name','category','brand','color','description','cabinet','shelf','date','status','notes']){if(form.elements[k])form.elements[k].value=x[k]??''}if(x.photo){document.getElementById('preview').src=x.photo;document.getElementById('preview').style.display='block'}}else form.elements.date.value=new Date().toISOString().slice(0,10)}
function editItem(id){openModal(id)}
function closeModal(){document.getElementById('modal').classList.remove('show');document.getElementById('form').reset();document.getElementById('preview').style.display='none';editingId=null}
function deleteEditing(){if(!editingId)return;if(confirm('Tem a certeza de que pretende excluir este registo?')){items=items.filter(x=>x.id!==editingId);save();renderAll();closeModal();toast('Registo excluído.')}}
function quickStatus(id,status){const x=items.find(i=>i.id===id);if(!x)return;x.status=status;save();renderAll();toast('Estado atualizado.')}
document.getElementById('photo').addEventListener('change',e=>{const f=e.target.files[0];if(!f)return;const r=new FileReader();r.onload=()=>{document.getElementById('preview').src=r.result;document.getElementById('preview').style.display='block'};r.readAsDataURL(f)});
document.getElementById('form').addEventListener('submit',e=>{e.preventDefault();const fd=new FormData(e.target);const f=document.getElementById('photo').files[0];const finish=photo=>{let x=editingId?items.find(i=>i.id===editingId):null;const id=editingId||'AP-'+String(Math.max(0,...items.map(x=>parseInt(x.id.slice(3))||0))+1).padStart(4,'0');const data={id,name:fd.get('name').trim(),category:fd.get('category'),brand:fd.get('brand').trim(),color:fd.get('color').trim(),description:fd.get('description').trim(),cabinet:+fd.get('cabinet'),shelf:+fd.get('shelf'),date:fd.get('date'),status:fd.get('status'),notes:fd.get('notes').trim(),photo:photo||x?.photo||''};if(x)Object.assign(x,data);else items.push(data);save();renderAll();closeModal();toast(editingId?'Objeto atualizado com sucesso.':'Objeto '+id+' cadastrado com sucesso.')};if(f){const r=new FileReader();r.onload=()=>finish(r.result);r.readAsDataURL(f)}else finish('')});

function formatDate(s){if(!s)return '—';return new Date(s+'T12:00:00').toLocaleDateString('pt-BR')}
function esc(s){return String(s||'').replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]))}
function toast(t){let x=document.getElementById('toast');x.textContent=t;x.classList.add('show');setTimeout(()=>x.classList.remove('show'),2600)}
